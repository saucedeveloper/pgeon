main type formula

function not  : formula -> formula
function and  : formula formula -> formula

function box : formula -> formula

rule rAxiom   : P ; not(P) ==X

rule rAnd     : and(P, Q)       ==> P ; Q
rule rNotNot  : not(not(P))     ==> P
rule rNotAnd  : not(and(P, Q))  ==> not(P) | not(Q)

rule rTheta   : P; Q ==> P

rule rT       : box(P) ==> box(P) ; P

tree rule rS4 : (not(box(P)); ...B) | ...T --> (not(P); ...B) | ...T
where {
  ...B : box(_)
}

main strategy prove : rTheta ; rS4 ; rNotNot; rAnd; rAxiom
