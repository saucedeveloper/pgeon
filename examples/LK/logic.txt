type formula
type term

function bot  : -> formula
function top  : -> formula

function not  : formula -> formula

function and  : formula formula -> formula
function or   : formula formula -> formula

binder forall: term. formula
binder exists: term. formula

rule ax1 : bot()      ==X
rule ax2 : not(top()) ==X
rule ax3 : P; not(P)  ==X

rule notnot : not(not(P))    ==> P
rule and : and(P, Q)         ==> P; Q
rule notor : not(or(P, Q))   ==> not(P); not(Q)
rule or : or(P, Q)           ==> P | Q
rule notand : not(and(P, Q)) ==> not(P) | not(Q)

rule exists : exists x. P ==> Q
where {
  Q = P[x <- @skolem()];
}
rule notforall : not(forall x. P) ==> not(Q)
where {
  Q = P[x <- @skolem()];
}

rule forall : forall x. P ==> Q
where {
  Q = P[x <- @fresh()];
}
rule notexists : not(exists x. P) ==> not(Q)
where {
  Q = P[x <- @fresh()];
}

rule ax4 : (P0 ; not(P1) ; B) | T --> T'
where {
  S = unify(P0, P1);
  T' = ((P0; not(P1); B) | T)[S];
}

strategy ax : ax1 || ax2 ||  ax4
strategy inv : ax || notnot || and || notor || or || notand
strategy notinv : exists || notforall || forall || notexists
strategy prove : ((ax *); (inv *); notinv) *
