
function inside stole lies honest : term -> formula
function accuses false_accusation : term term -> formula

function alice bob carol : -> term

inside(alice());
inside(bob());
inside(carol());

forall x. impl(stole(x), inside(x));

forall x. impl(stole(x), lies(x));

forall x. impl(honest(x), not(lies(x)));

forall x. impl(inside(x), or(honest(x), lies(x)));

accuses(bob(), carol());
accuses(carol(), bob());

forall y. not(accuses(alice(), y));

forall x. forall y. impl(false_accusation(x,y), lies(x));

forall x. forall y. impl(and(accuses(x,y), not(stole(y))), false_accusation(x,y));

forall x. forall y. impl(and(accuses(x,y), honest(x)), stole(y));

exists x. and(inside(x), honest(x));

not(stole(bob()));
